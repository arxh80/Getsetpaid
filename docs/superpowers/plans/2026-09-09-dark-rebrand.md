# Dark Rebrand Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the light `#FFF4F4`/two-blue palette with a dark ground built on the new logo's `#20FC8F`, and replace the type-only wordmark with the real traced logo lockup.

**Architecture:** Every colour in this app already resolves through a Tailwind v4 `@theme` semantic token (`bg-surface`, `text-ink-muted`, `border-rule`), so the palette change is a token-value edit plus a handful of base rules — no component class changes. A new source-reading unit test locks `app/globals.css` and `scripts/contrast.py` to the same values so the two can never drift. The logo becomes an inline SVG generated by a committed tracer script, replacing `components/wordmark.tsx` in place so both existing call sites are untouched.

**Tech Stack:** Next.js 15 App Router, Tailwind CSS v4 (CSS-first `@theme`, no `tailwind.config.js`), Vitest (node environment, `.test.ts` only), Playwright + axe-core, Python 3 + Pillow for the tracer.

```ts
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { describe, expect, it } from "vitest";

// WHY THIS TEST EXISTS.
//
// The palette lives in two places by necessity: `app/globals.css`'s @theme
// block is what the browser gets, and `scripts/contrast.py`'s TOKENS dict is
// what the WCAG audit measures. Neither can import the other -- one is CSS
// consumed by Tailwind's compiler, the other is Python. So the audit can pass
// while the shipped stylesheet says something different, which is the exact
// failure mode that makes a "contrast script" worthless: a green tick
// measuring values nobody serves.
//
// This test is the seam. It parses both files and asserts they agree.

const root = fileURLToPath(new URL("../../../", import.meta.url));

function readThemeColors(): Map<string, string> {
  const css = readFileSync(`${root}app/globals.css`, "utf8");
  const theme = css.slice(css.indexOf("@theme {"));
  const found = new Map<string, string>();
  // Only `--color-*: #hex;` -- rgb() values (scrim) and non-colour tokens are
  // out of scope for a hex-pair audit and are asserted separately below.
  for (const [, name, hex] of theme.matchAll(
    /--color-([a-z0-9-]+):\s*(#[0-9A-Fa-f]{6})\s*;/g,
  )) {
    found.set(name, hex.toUpperCase());
  }
  return found;
}

function readContrastTokens(): Map<string, string> {
  const py = readFileSync(`${root}scripts/contrast.py`, "utf8");
  const block = py.slice(py.indexOf("TOKENS = {"), py.indexOf("\n}", py.indexOf("TOKENS = {")));
  const found = new Map<string, string>();
  for (const [, name, hex] of block.matchAll(
    /"([a-z0-9-]+)":\s*"(#[0-9A-Fa-f]{6})"/g,
  )) {
    found.set(name, hex.toUpperCase());
  }
  return found;
}

// The spec's § 3.1 table, transcribed. Written out rather than derived from
// either file so a wrong edit to BOTH files still fails here.
const EXPECTED: Record<string, string> = {
  shell: "#0B0B0E",
  surface: "#15151A",
  "surface-raised": "#1E1E25",
  ink: "#F5F5F7",
  "ink-muted": "#9A9AA4",
  primary: "#20FC8F",
  "on-primary": "#0B0B0E",
  accent: "#20FC8F",
  rule: "#2A2A33",
  "rule-strong": "#6A6A78",
  success: "#20FC8F",
  warning: "#FFB020",
  danger: "#FF6B61",
  focus: "#F5F5F7",
  "focus-on-primary": "#0B0B0E",
};

describe("design token contract", () => {
  it("app/globals.css carries exactly the specified palette", () => {
    const theme = readThemeColors();
    for (const [name, hex] of Object.entries(EXPECTED)) {
      expect(theme.get(name), `--color-${name}`).toBe(hex);
    }
  });

  it("scripts/contrast.py measures the same values globals.css serves", () => {
    const theme = readThemeColors();
    const py = readContrastTokens();
    for (const [name, hex] of py) {
      expect(theme.get(name), `--color-${name} in globals.css`).toBe(hex);
    }
  });

  it("the deleted light-palette tokens are gone", () => {
    const theme = readThemeColors();
    expect(theme.has("lilac")).toBe(false);
    expect(theme.has("pink")).toBe(false);
  });

  it("no removed hex survives anywhere in the theme", () => {
    const css = readFileSync(`${root}app/globals.css`, "utf8");
    for (const dead of [
      "#FFF4F4",
      "#14141C",
      "#0062FF",

    }
  });
});
```

Create `scripts/trace-logo.py`:

```python
"""Trace brand/*.png into SVG path data for components/wordmark.tsx.

    python scripts/trace-logo.py

Writes components/wordmark.paths.ts. Re-run after changing the artwork.

WHY THIS EXISTS RATHER THAN A ONE-LINE potrace CALL: no vector tracer is
installed on this project's machines -- potrace, vtracer, inkscape and
ImageMagick's `magick` are all absent -- and numpy is not available to Python
either. Pillow is. So this is marching squares plus Douglas-Peucker in pure
Python, which is a hundred lines and removes a build dependency nobody has.

The artwork is flat single-colour with hard edges, which is the easy case for
a contour tracer: no anti-aliasing decisions beyond one alpha threshold, no
colour quantisation, no curve fitting. Straight line segments are enough --
the output is polygons, not beziers, and at the sizes this renders (20px in a
header, 200px on a marketing page) the difference is not visible.
"""

import sys
from pathlib import Path

from PIL import Image

ROOT = Path(__file__).resolve().parent.parent
OUT = ROOT / "components" / "wordmark.paths.ts"

# Downsample before tracing. The sources are 2226px and 4172px wide; contour
# detail at that resolution is pixel noise once the result is normalised to a
# 100-unit viewBox, and it costs a hundredfold in points.
TRACE_WIDTH = 600

# Douglas-Peucker tolerance, in traced-pixel units. 0.8 keeps the outline's
# corners sharp (they are real corners in the artwork, not sampling artefacts)
# while collapsing the staircase along every straight edge.
EPSILON = 0.8


def load_mask(path, mode, box=None):
    """Return a (w, h) bool grid: True where the artwork is.

    mode "alpha"  -- opaque pixels (the mark, on transparency)
    mode "light"  -- bright pixels (the wordmark, white on black)
    """
    im = Image.open(path).convert("RGBA")
    if box:
        im = im.crop(box)
    w = TRACE_WIDTH
    h = max(1, round(im.height * TRACE_WIDTH / im.width))
    im = im.resize((w, h), Image.LANCZOS)
    px = im.load()
    grid = [[False] * w for _ in range(h)]
    for y in range(h):
        for x in range(w):
            r, g, b, a = px[x, y]
            if mode == "alpha":
                grid[y][x] = a >= 128
            else:
                grid[y][x] = a >= 128 and (r + g + b) / 3 >= 128
    return grid, w, h


def contours(grid, w, h):
    """Marching squares over the cell corners, returning closed point loops.

    Walks every unvisited boundary edge between a True and a False cell and
    follows it until it returns to the start. Produces both outer contours and
    the inner contours of holes -- which is exactly what an OUTLINED mark
    needs, since every stroke has two edges.
    """
    def on(x, y):
        return 0 <= x < w and 0 <= y < h and grid[y][x]

    # Edge key: (x, y, direction). Direction 0=up,1=right,2=down,3=left as the
    # boundary is walked with the filled region on the left.
    seen = set()
    loops = []
    # Direction vectors and the two cells that decide the next turn.
    for sy in range(h):
        for sx in range(w):
            if not on(sx, sy) or on(sx, sy - 1):
                continue  # not a top edge of a filled run
            if (sx, sy, 1) in seen:
                continue
            loop = []
            x, y, d = sx, sy, 1  # start heading right along the top edge
            while True:
                if (x, y, d) in seen:
                    break
                seen.add((x, y, d))
                if d == 1:
                    loop.append((x, y))
                    nxt = (x + 1, y)
                    if on(x, y - 1):
                        x, y, d = x, y - 1, 0
                    elif on(x, y):
                        x, y, d = nxt[0], nxt[1], 1
                    else:
                        d = 2
                elif d == 2:
                    loop.append((x + 1, y))
                    if on(x, y):
                        x, y, d = x, y + 1, 2
                    else:
                        d = 3
                elif d == 3:
                    loop.append((x + 1, y + 1))
                    if on(x, y):
                        x, y, d = x - 1, y, 3
                    else:
                        d = 0
                else:
                    loop.append((x, y + 1))
                    if on(x, y):
                        x, y, d = x, y - 1, 0
                    else:
                        d = 1
                if (x, y, d) == (sx, sy, 1):
                    break
            if len(loop) >= 8:
                loops.append(loop)
    return loops


def perpendicular_distance(p, a, b):
    (px, py), (ax, ay), (bx, by) = p, a, b
    dx, dy = bx - ax, by - ay
    if dx == 0 and dy == 0:
        return ((px - ax) ** 2 + (py - ay) ** 2) ** 0.5
    return abs(dy * px - dx * py + bx * ay - by * ax) / (dx * dx + dy * dy) ** 0.5


def simplify(points, epsilon):
    """Douglas-Peucker, iterative so a long contour cannot blow the stack."""
    if len(points) < 3:
        return points
    keep = [False] * len(points)
    keep[0] = keep[-1] = True
    stack = [(0, len(points) - 1)]
    while stack:
        lo, hi = stack.pop()
        worst, worst_i = 0.0, -1
        for i in range(lo + 1, hi):
            d = perpendicular_distance(points[i], points[lo], points[hi])
            if d > worst:
                worst, worst_i = d, i
        if worst > epsilon:
            keep[worst_i] = True
            stack.append((lo, worst_i))
            stack.append((worst_i, hi))
    return [p for p, k in zip(points, keep) if k]


def to_path(loops, w, h, scale_to=100.0):
    """Normalise to a viewBox whose HEIGHT is `scale_to`, and emit `d`."""
    s = scale_to / h
    parts = []
    for loop in loops:
        pts = simplify(loop, EPSILON)
        if len(pts) < 3:
            continue
        d = " ".join(
            f"{'M' if i == 0 else 'L'}{round(x * s, 1)} {round(y * s, 1)}"
            for i, (x, y) in enumerate(pts)
        )
        parts.append(d + "Z")
    return "".join(parts), round(w * s, 1), scale_to


def find_wordmark_box(path):
    """Bounding box of the WHITE pixels only, so the green mark is excluded."""
    im = Image.open(path).convert("RGBA")
    px = im.load()
    minx, miny, maxx, maxy = im.width, im.height, -1, -1
    for y in range(im.height):
        for x in range(im.width):
            r, g, b, a = px[x, y]
            if a >= 128 and r >= 200 and g >= 200 and b >= 200:
                minx, miny = min(minx, x), min(miny, y)
                maxx, maxy = max(maxx, x), max(maxy, y)
    if maxx < 0:
        sys.exit("no white pixels found in " + str(path))
    return (minx, miny, maxx + 1, maxy + 1)


def main():
    mark_grid, mw, mh = load_mask(ROOT / "brand" / "LOGO.png", "alpha")
    mark_d, mark_w, mark_h = to_path(contours(mark_grid, mw, mh), mw, mh)

    word_box = find_wordmark_box(ROOT / "brand" / "logo_with_text.png")
    word_grid, ww, wh = load_mask(
        ROOT / "brand" / "logo_with_text.png", "light", word_box
    )
    word_d, word_w, word_h = to_path(contours(word_grid, ww, wh), ww, wh)

    OUT.write_text(
        "// GENERATED FILE -- do not edit.\n"
        "// Regenerate with: python scripts/trace-logo.py\n"
        "// Source artwork: brand/LOGO.png, brand/logo_with_text.png\n"
        "\n"
        f'export const MARK_PATH =\n  "{mark_d}";\n'
        f'export const MARK_VIEWBOX = "0 0 {mark_w} {mark_h}";\n'
        "\n"
        f'export const WORD_PATH =\n  "{word_d}";\n'
        f'export const WORD_VIEWBOX = "0 0 {word_w} {word_h}";\n',
        encoding="utf-8",
    )
    total = len(mark_d) + len(word_d)
    print(f"mark  {mark_w}x{mark_h}  {len(mark_d)} bytes")
    print(f"word  {word_w}x{word_h}  {len(word_d)} bytes")
    print(f"total {total} bytes (budget 15360)")
    if total >= 15360:
        sys.exit("over the path budget -- raise EPSILON and re-run")


if __name__ == "__main__":
    main()
```
